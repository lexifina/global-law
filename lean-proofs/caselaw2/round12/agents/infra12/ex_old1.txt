                        excluded_keys.add((d["table"], str(d["by_node"]), str(d["target_node"]), d["kind2"]))
            for cname in ("decoy_id", "row2_id"):

                    dc = kr[0].index(cname)
                    excluded |= {p[dc] for p in kr[1:] if len(p) > dc and p[dc]}
